145std::istream& operator>>(std::istream& in, Studentas& s) {
146 std::string vardas, pavarde;
147 in >> vardas >> pavarde;
148
149 if (!in) return in;
150
151 s.setVardas(vardas);
152 s.setPavarde(pavarde);
153
154 Vector<short int> nd;
155 int x = 0;
156
157 while (in >> x) {
158 nd.push_back(static_cast<short int>(x));
159 if (in.peek() == '\n') break;
160 }
161
162 if (!nd.empty()) {
163 s.setEgzaminas(static_cast<uint8_t>(nd.back()));
164 nd.pop_back();
165
166 int suma = 0;
167 for (short int v : nd) {
168 suma += v;
169 }
170
171 std::sort(nd.begin(), nd.end());
172
173 const std::size_t n = nd.size();
174 double med = 0.0;
175
176 if (n > 0) {
177 if (n % 2 == 0) {
178 med = (nd[n / 2 - 1] + nd[n / 2]) / 2.0;
179 } else {
180 med = nd[n / 2];
181 }
182 }
183
184 s.setNamuDarbaiVid100(
185 (n > 0)
186 ? static_cast<uint16_t>((static_cast<double>(suma) / static_cast<double>(n)) * 100.0)
187 : 0
188 );
189
190 s.setNamuDarbaiMed100(static_cast<uint16_t>(med * 100.0));
191 s.setND(nd);
192 }
193
194 return in;
195}
